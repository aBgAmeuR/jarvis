# Colors

All colors are CSS variables defined in `theme.css` and exposed as Tailwind utilities (`bg-primary`, `text-muted-foreground`). Each has a dark value; toggle the theme to compare.

export const Swatch = ({ name, note }) => (
  <div className="flex flex-col gap-1.5">
    <div className={`bg-${name} border-border h-12 rounded-md border`} />
    <code className="text-xs">{name}</code>
    {note && <span className="text-muted-foreground text-xs">{note}</span>}
  </div>
);

export const Grid = ({ children }) => (
  <div className="my-4 grid grid-cols-2 gap-4 sm:grid-cols-4">{children}</div>
);
